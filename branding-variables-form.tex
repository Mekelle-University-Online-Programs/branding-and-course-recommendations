\section*{Appendix --- Python Contrast Verification Script}

Save as \variable{scripts/check\_contrast.py}. Run:
\texttt{python3 scripts/check\_contrast.py branding.json}

\begin{lstlisting}[language=Python]
#!/usr/bin/env python3
import json, sys
AA_NORMAL, AA_LARGE, AA_NON_TEXT = 4.5, 3.0, 3.0

def hex_to_rgb(h):
    h = h.lstrip("#")
    return tuple(int(h[i:i+2], 16) / 255.0 for i in (0, 2, 4))

def relative_luminance(rgb):
    def channel(c):
        return c / 12.92 if c <= 0.03928 else ((c + 0.055) / 1.055) ** 2.4
    r, g, b = (channel(c) for c in rgb)
    return 0.2126 * r + 0.7152 * g + 0.0722 * b

def ratio(fg, bg):
    l1 = relative_luminance(hex_to_rgb(fg))
    l2 = relative_luminance(hex_to_rgb(bg))
    lighter, darker = max(l1, l2), min(l1, l2)
    return (lighter + 0.05) / (darker + 0.05)

CHECKS = [
    ("TEXT_COLOR",       "#FFFFFF", AA_NORMAL,   "Body text on white"),
    ("TEXT_COLOR",       "#F5F7FA", AA_NORMAL,   "Body text on note bg"),
    ("LINK_COLOR",       "#FFFFFF", AA_NORMAL,   "Link on white"),
    ("LINK_COLOR",       "#F5F7FA", AA_NORMAL,   "Link on callout"),
    ("PRIMARY_COLOR",    "#FFFFFF", AA_NORMAL,   "Heading on white"),
    ("PRIMARY_COLOR",    "#F5F7FA", AA_NORMAL,   "Heading on callout"),
    ("PRIMARY_COLOR",    "#FFFFFF", AA_LARGE,    "Table header text"),
    ("MUTED_TEXT_COLOR", "#FFFFFF", AA_NORMAL,   "Caption text"),
    ("ACCENT_COLOR",     "#FFFFFF", AA_NON_TEXT, "Focus outline"),
    ("WARNING_COLOR",    "#FFFFFF", AA_NON_TEXT, "Warning border"),
    ("SUCCESS_COLOR",    "#FFFFFF", AA_NON_TEXT, "Success border"),
    ("SECONDARY_COLOR",  "#F5F7FA", AA_NON_TEXT, "Definition callout"),
]

def main(path):
    with open(path) as f:
        values = json.load(f)
    failures = 0
    print(f"{'Check':<40}{'Ratio':>8}  {'Min':>5}  Result")
    print("-" * 68)
    for key, bg, minimum, label in CHECKS:
        fg = values.get(key)
        if not fg:
            print(f"{label:<40}{'--':>8}  {minimum:>5}  MISSING {key}")
            failures += 1
            continue
        r = ratio(fg, bg)
        ok = r >= minimum
        if not ok:
            failures += 1
        print(f"{label:<40}{r:>8.2f}  {minimum:>5}  {'OK' if ok else 'FAIL'}")
    print("-" * 68)
    print(f"Total failures: {failures}")
    sys.exit(1 if failures else 0)

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python3 check_contrast.py branding.json")
        sys.exit(2)
    main(sys.argv[1])
\end{lstlisting}
